# 统一文件资源系统 —— 实现规格

**定位**：内容寻址的多模态文件存储服务。为通用消息模型及其他业务提供统一的文件持久化、去重、查询、删除能力。

当前仅支持：**图片**（JPEG / PNG / WebP）、**PDF**。

---

## 设计原则

1. **内容寻址，自动去重**：以文件二进制内容的 SHA-256 哈希为去重键。相同内容的文件无论上传多少次，只存一份，返回同一个 resourceId。
2. **双存储**：每个文件同时拥有本地路径和阿里云国际 OSS URL。本地为热缓存 / 离线使用，OSS 为持久化 / 跨端共享。
3. **服务内聚**：文件读取、元数据解析、本地写入、OSS 上传均在服务内部完成，调用方只需提供文件路径或 base64。
4. **所有接口支持批量**：每个 API 同时接受单条和批量输入，批量有上限（默认 `BATCH_LIMIT = 20`）。
5. **消息模型集成**：通用消息模型中 user / tool 消息的 `ImagePart` / `PdfPart` 通过 `resourceId` 引用本系统。当前 Claude / OpenAI / Gemini 的 assistant 响应不含多模态数据（仅 text / thinking / tool_use），raw 原样存储无需处理。

## 一、类型定义（`schema.ts`）

```ts
export type FileCategory = "image" | "pdf";

export type MimeType =
  | "image/jpeg" | "image/png" | "image/webp"
  | "application/pdf";

export interface FileResource {
  /** 资源唯一 ID（UUID v4） */
  id: string;

  /** 文件二进制内容的 SHA-256 哈希（十六进制小写，64 字符），去重键 */
  contentHash: string;

  /** 文件类别 */
  category: FileCategory;

  /** MIME 类型 */
  mimeType: MimeType;

  /** 文件大小（字节） */
  size: number;

  /** 本地文件路径 */
  localPath: string;

  /** 阿里云国际版 OSS URL */
  ossUrl: string;

  /** 类别相关的元数据 */
  metadata: ImageMetadata | PdfMetadata;

  /** 创建时间 */
  createdAt: string;   // ISO 8601
}

export interface ImageMetadata {
  category: "image";
  width: number;       // 像素
  height: number;      // 像素
}

export interface PdfMetadata {
  category: "pdf";
  pageCount: number;   // 总页数
}
```

## 二、对外 API（`resource-service.ts`）

所有接口由 `ResourceService` 统一提供。底层依赖 `ResourceStore`（DB 持久化）和 `OssUploader`（OSS 上传）两个注入实现。

```ts
import type { FileResource, FileCategory, MimeType } from "./schema";

/** 单次批量上限 */
const BATCH_LIMIT = 20;

export class ResourceService {
  constructor(
    private store: ResourceStore,
    private oss: OssUploader,
    private config: { localStorageRoot: string },
  ) {}

  // ──────────────────────────────────────────────
  //  API 1：从本地文件路径上传
  // ──────────────────────────────────────────────

  /**
   * 传入本地文件路径，系统读取文件、解析元数据、去重、存储、上传 OSS。
   * 返回 { id, ossUrl } 数组（与输入顺序一一对应）。
   *
   * 若文件内容与已有资源相同（SHA-256 匹配），返回已有资源的 id 和 url。
   */
  async uploadFromFiles(
    items: { filePath: string; mimeType: MimeType }[],
  ): Promise<{ id: string; ossUrl: string }[]> {
    this.assertBatchLimit(items.length);
    return Promise.all(items.map(item => this.ingestFromFile(item.filePath, item.mimeType)));
  }

  // ──────────────────────────────────────────────
  //  API 2：从 base64 上传
  // ──────────────────────────────────────────────

  /**
   * 传入 base64 编码内容 + 文件类型信息，系统解码、解析元数据、去重、本地存储、上传 OSS。
   * 返回 { id, localPath, ossUrl } 数组。
   */
  async uploadFromBase64(
    items: { base64: string; mimeType: MimeType }[],
  ): Promise<{ id: string; localPath: string; ossUrl: string }[]> {
    this.assertBatchLimit(items.length);
    return Promise.all(items.map(item => this.ingestFromBase64(item.base64, item.mimeType)));
  }

  // ──────────────────────────────────────────────
  //  API 3：查询资源
  // ──────────────────────────────────────────────

  /**
   * 根据资源 ID 批量获取完整信息。
   * 返回 { id, localPath, base64, mimeType, ossUrl, metadata } 数组。
   * 找不到的 id 对应位置返回 null。
   */
  async getResources(
    ids: string[],
  ): Promise<(ResourceInfo | null)[]> {
    this.assertBatchLimit(ids.length);
    return Promise.all(ids.map(id => this.resolveResource(id)));
  }

  // ──────────────────────────────────────────────
  //  API 4：删除资源
  // ──────────────────────────────────────────────

  /**
   * 根据资源 ID 批量删除。删除本地文件 + OSS 对象 + DB 记录。
   * 不存在的 id 静默跳过。
   */
  async deleteResources(ids: string[]): Promise<void> {
    this.assertBatchLimit(ids.length);
    await Promise.all(ids.map(id => this.removeResource(id)));
  }

  // ──────────────────────────────────────────────
  //  内部实现
  // ──────────────────────────────────────────────

  private assertBatchLimit(count: number): void {
    if (count > BATCH_LIMIT) throw new Error(`Batch limit exceeded: ${count} > ${BATCH_LIMIT}`);
    if (count === 0) throw new Error("Empty batch");
  }

  /** 从本地文件路径入库 */
  private async ingestFromFile(
    filePath: string, mimeType: MimeType,
  ): Promise<{ id: string; ossUrl: string }> {
    const content = await fs.readFile(filePath);
    const resource = await this.ingestContent(Buffer.from(content), mimeType);
    return { id: resource.id, ossUrl: resource.ossUrl };
  }

  /** 从 base64 入库 */
  private async ingestFromBase64(
    base64: string, mimeType: MimeType,
  ): Promise<{ id: string; localPath: string; ossUrl: string }> {
    const content = Buffer.from(base64, "base64");
    const resource = await this.ingestContent(content, mimeType);
    return { id: resource.id, localPath: resource.localPath, ossUrl: resource.ossUrl };
  }

  /** 核心入库逻辑：hash → 去重 → 解析元数据 → 本地存储 → OSS → DB */
  private async ingestContent(content: Buffer, mimeType: MimeType): Promise<FileResource> {
    const contentHash = createHash("sha256").update(content).digest("hex");

    // 去重
    const existing = await this.store.getByHash(contentHash);
    if (existing) return existing;

    // 解析元数据（服务内部完成，调用方无需关心）
    const category = mimeType.startsWith("image/") ? "image" : "pdf";
    const metadata = category === "image"
      ? await parseImageMeta(content)           // → { category:"image", width, height }
      : await parsePdfMeta(content);            // → { category:"pdf", pageCount }

    // 生成 ID、写本地（hash 分片目录）、传 OSS
    const id = crypto.randomUUID();
    const localPath = buildLocalPath(this.config.localStorageRoot, category, contentHash, mimeType);
    await saveLocal(localPath, content);

    const ossUrl = await this.oss.upload(content, id, mimeType);

    const resource: FileResource = {
      id, contentHash, category, mimeType,
      size: content.byteLength,
      localPath, ossUrl, metadata,
      createdAt: new Date().toISOString(),
    };
    await this.store.save(resource);
    return resource;
  }

  /** 查询单个资源，返回完整信息含 base64 */
  private async resolveResource(id: string): Promise<ResourceInfo | null> {
    const resource = await this.store.getById(id);
    if (!resource) return null;

    const content = await fs.readFile(resource.localPath);
    return {
      id: resource.id,
      localPath: resource.localPath,
      base64: content.toString("base64"),
      mimeType: resource.mimeType,
      ossUrl: resource.ossUrl,
      metadata: resource.metadata,
    };
  }

  /** 删除单个资源 */
  private async removeResource(id: string): Promise<void> {
    const resource = await this.store.getById(id);
    if (!resource) return;

    // 并行删除：本地文件 + OSS + DB
    await Promise.all([
      fs.unlink(resource.localPath).catch(() => {}),
      this.oss.delete(resource.id).catch(() => {}),
      this.store.delete(resource.id),
    ]);
  }
}

/** 查询返回的资源完整信息 */
export interface ResourceInfo {
  id: string;
  localPath: string;
  base64: string;
  mimeType: MimeType;
  ossUrl: string;
  metadata: ImageMetadata | PdfMetadata;
}
```

## 四、本地文件存储设计

### 4.1 目录结构

采用 **contentHash 分片** 方案（类似 Git 对象存储）。用哈希前 2 字符做子目录，避免单目录文件过多：

```
{localStorageRoot}/
├── image/
│   ├── a1/
│   │   ├── a1b2c3d4e5...64chars.jpg
│   │   └── a1ff09c8e2...64chars.png
│   ├── 3e/
│   │   └── 3e8a02bd71...64chars.webp
│   └── .../                          // 最多 256 个子目录 (00-ff)
└── pdf/
    ├── 7c/
    │   └── 7c4f2a91be...64chars.pdf
    └── .../
```

**命名规则**：`{localStorageRoot}/{category}/{hash[0:2]}/{hash}.{ext}`

- `category`：`image` 或 `pdf`，一级分类
- `hash[0:2]`：contentHash 前 2 字符，二级分片（256 个桶）
- `hash.ext`：完整 64 字符哈希 + 扩展名

### 4.2 设计决策

| 决策 | 选择 | 理由 |
|---|---|---|
| 文件名 | contentHash（非 UUID） | 文件系统自描述，重复写入幂等，可离线校验完整性 |
| 分片策略 | hash 前 2 字符 | 均匀分布到 256 个子目录，单桶万级文件无压力 |
| 扩展名 | 保留 `.jpg/.png/.pdf` 等 | 便于 OS 预览、调试、第三方工具读取 |
| 写入幂等 | 先检查文件是否存在再写 | contentHash 相同 = 内容相同，跳过写入 |

### 4.3 路径生成与写入

```ts
import { createHash } from "crypto";
import * as fs from "fs/promises";
import * as path from "path";

/** MIME → 文件扩展名 */
function mimeToExt(mime: MimeType): string {
  const map: Record<MimeType, string> = {
    "image/jpeg": "jpg",
    "image/png": "png",
    "image/webp": "webp",
    "application/pdf": "pdf",
  };
  return map[mime] ?? "bin";
}

/** 根据 contentHash 生成本地文件路径 */
function buildLocalPath(
  root: string, category: FileCategory, contentHash: string, mimeType: MimeType,
): string {
  const shard = contentHash.slice(0, 2);           // 前 2 字符做子目录
  const ext = mimeToExt(mimeType);
  return path.join(root, category, shard, `${contentHash}.${ext}`);
}

/** 写入本地文件（幂等：同 hash 文件已存在则跳过） */
async function saveLocal(localPath: string, content: Buffer): Promise<void> {
  try {
    await fs.access(localPath);
    // 文件已存在，跳过写入（contentHash 相同 = 内容相同）
    return;
  } catch {
    // 文件不存在，正常写入
  }
  await fs.mkdir(path.dirname(localPath), { recursive: true });
  // 先写临时文件再 rename，防止写入中断导致损坏文件
  const tmpPath = `${localPath}.tmp.${process.pid}`;
  await fs.writeFile(tmpPath, content);
  await fs.rename(tmpPath, localPath);
}
```

### 4.4 元数据解析

```ts
/** 从图片二进制解析宽高 */
async function parseImageMeta(content: Buffer): Promise<ImageMetadata> {
  const { width, height } = await sharp(content).metadata();  // 或其他库
  return { category: "image", width: width!, height: height! };
}

/** 从 PDF 二进制解析页数 */
async function parsePdfMeta(content: Buffer): Promise<PdfMetadata> {
  const doc = await PDFDocument.load(content);                // pdf-lib
  return { category: "pdf", pageCount: doc.getPageCount() };
}
```

## 五、API 调用流程图

```
┌──────────── API 1: uploadFromFiles ────────────┐
│  filePath + mimeType                           │
│  → fs.readFile(filePath)                       │
│  → SHA-256(content)                            │
│  → store.getByHash → 命中 → 返回已有 {id, url} │
│  →                 → 未命中 ↓                   │
│     parseImageMeta / parsePdfMeta              │
│     → saveLocal → oss.upload → store.save      │
│     → 返回新 {id, url}                         │
└────────────────────────────────────────────────┘

┌──────────── API 2: uploadFromBase64 ───────────┐
│  base64 + mimeType                             │
│  → Buffer.from(base64, "base64")               │
│  → 同上 ingestContent 流程                      │
│  → 返回 {id, localPath, url}                   │
└────────────────────────────────────────────────┘

┌──────────── API 3: getResources ───────────────┐
│  ids[]                                         │
│  → store.getById → fs.readFile → base64 编码    │
│  → 返回 {id, localPath, base64, mimeType, url, │
│          metadata}                             │
└────────────────────────────────────────────────┘

┌──────────── API 4: deleteResources ────────────┐
│  ids[]                                         │
│  → store.getById → 并行:                        │
│     fs.unlink + oss.delete + store.delete      │
└────────────────────────────────────────────────┘
```

## 六、与通用消息模型的集成

### 6.1 消息中的多模态引用

user / tool 消息的多模态数据通过 resourceId 引用本系统：

- **user / tool**：`ImagePart.resourceId` / `PdfPart.resourceId` 直接存储。
- **assistant**：当前各平台响应不含多模态（仅 text / thinking / tool_use），`providerExtras.*.raw` 原样存储，不涉及资源系统。

### 6.2 消息模型调用本系统的时机

| 时机 | 调用的 API | 说明 |
|---|---|---|
| 用户上传附件 | `uploadFromFiles` | 用户选择本地文件 → 获取 resourceId |
| ~~`ingestResponse` 脱水~~ | ~~N/A~~ | 当前 assistant 响应无多模态，无需处理 |
| `renderMessages` 注水 | `getResources` | 回灌 raw 或构造请求时 → resourceId → base64 |
| `renderMessages` user 消息 | `getResources` | ImagePart/PdfPart 的 resourceId → base64/url |

### 6.3 消息模型侧的接口适配

消息模型不直接依赖 `ResourceService` class，而是通过 `ResourceResolver` 接口解耦：

```ts
// 渲染时使用（renderMessages 接收）
interface ResourceResolver {
  getMeta(id: string): Promise<{ mimeType: MimeType; metadata: ImageMetadata | PdfMetadata }>;
  getBase64(id: string): Promise<string>;
  getUrl(id: string): Promise<string>;
}
// 实现：基于 service.getResources 封装
```

## 七、完整使用示例

```ts
const service = new ResourceService(store, oss, { localStorageRoot: "./data/resources" });

// ---- 用户上传文件 ----
const [imgRes, pdfRes] = await service.uploadFromFiles([
  { filePath: "/tmp/photo.jpg", mimeType: "image/jpeg" },
  { filePath: "/tmp/report.pdf", mimeType: "application/pdf" },
]);
// imgRes = { id: "abc-123", ossUrl: "https://..." }
// pdfRes = { id: "def-456", ossUrl: "https://..." }

// ---- 构造消息 ----
const pdfMeta = await service.getResources([pdfRes.id]);
addUserMessage(conv, "claude", [
  { kind: "text", text: "请分析这份报告" },
  { kind: "image", resourceId: imgRes.id },
  { kind: "pdf", resourceId: pdfRes.id },
], [
  { kind: "text" },
  { kind: "image" },
  { kind: "pdf", pageCount: (pdfMeta[0]!.metadata as PdfMetadata).pageCount },
]);

// ---- LLM 响应入库（脱水） ----
const uploadFn = async (buf: Buffer, _cat: string, mime: string) => {
  const [r] = await service.uploadFromBase64([{ base64: buf.toString("base64"), mimeType: mime as MimeType }]);
  return r.id;
};
await ingestResponse(conv, "claude", rawResp, uploadFn);

// ---- 渲染请求（注水） ----
const resolver: ResourceResolver = {
  async getMeta(id) { const [r] = await service.getResources([id]); return r!; },
  async getBase64(id) { const [r] = await service.getResources([id]); return r!.base64; },
  async getUrl(id) { const [r] = await service.getResources([id]); return r!.ossUrl; },
};
const rendered = await renderMessages(conv, "claude", resolver);
```

## 八、纪律清单

1. **去重靠 SHA-256 哈希**，不靠文件名、路径、大小。相同内容 = 相同资源 = 相同 id。
2. **元数据由服务内部解析**：图片宽高、PDF 页数在 `ingestContent` 中自动完成，调用方无需关心。
3. **本层不做格式转换**：不压缩图片、不转格式。调用方在上传前处理好。
4. **批量有上限**：所有 API 限制 `BATCH_LIMIT = 20`，超出直接抛错。调用方如需处理更多，自行分批。
5. **删除是物理删除**：本地文件 + OSS + DB 记录一并删除。不做软删除。如需回收站机制由调用方在上层实现。
6. **查询返回 base64 是懒加载**：`getResources` 每次从本地文件读取并编码，不缓存 base64。调用方应按需调用，避免批量获取大文件。
